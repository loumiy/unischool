// Area 2, review III: a newcomer's first year, counted. Founds a college in
// headless Chromium against the production build, follows the walkthrough,
// then the chapters (opening each drawer the game opens, pressing its first
// Show me and the ringed control it leads to), answers every letter with its
// advance button, and plays the first summer (the Review, the admissions page,
// the students). At every new thing on screen it records the words of the coach
// card, the letter, the drawer, the ticker strip and the whole page, and takes a
// screenshot at 1440x900 (and the drawers again at 1280x720).
//
//   node opening.mjs <out-dir> [--years=1]
import { mkdirSync, writeFileSync } from 'node:fs';
import { join } from 'node:path';
import { createRequire } from 'node:module';
const require = createRequire('/home/claude/loumiy/unischool/unischool/package.json');
const { chromium } = require('playwright-core');

const OUT = process.argv[2] ?? 'opening-out';
const YEARS = Number(process.argv.find((a) => a.startsWith('--years='))?.slice(8) ?? '1');
const URL = process.env.CAMPUS_URL ?? 'http://localhost:4102/';
const TIMEOUT_S = Number(process.env.TIMEOUT_S ?? 900);
mkdirSync(OUT, { recursive: true });

const browser = await chromium.launch({ executablePath: '/opt/pw-browsers/chromium-1194/chrome-linux/chrome', args: ['--no-sandbox'] });
const context = await browser.newContext({ viewport: { width: 1440, height: 900 } });
await context.addInitScript(() => {
  // Statistics answered, so the notice does not sit over the opening.
  if (!localStorage.getItem('unischool.settings.v1')) localStorage.setItem('unischool.settings.v1', JSON.stringify({ stats: 'off' }));
});
const page = await context.newPage();
const errors = [];
page.on('pageerror', (e) => errors.push(String(e)));
await page.goto(URL, { waitUntil: 'networkidle' });
await page.waitForTimeout(800);

const started = Date.now();
const t = () => ((Date.now() - started) / 1000).toFixed(0);
const rows = [];
let n = 0;
const seen = new Set();

async function words(sel) {
  return page.evaluate((sel) => {
    const vis = (el) => el.getClientRects().length > 0 && getComputedStyle(el).visibility !== 'hidden';
    const els = [...document.querySelectorAll(sel)].filter(vis);
    if (!els.length) return null;
    const txt = els.map((e) => e.innerText).join(' ');
    return { words: txt.split(/\s+/).filter((w) => /[A-Za-z0-9$]/.test(w)).length, text: txt.replace(/\s+/g, ' ').trim() };
  }, sel);
}
async function clockNow() {
  const year = (await page.locator('.toolbar-clock').first().innerText({ timeout: 300 }).catch(() => '')).match(/Year (\d+)/)?.[1];
  const week = (await page.locator('.toolbar-calendar-week').first().innerText({ timeout: 300 }).catch(() => '')).trim();
  return year ? `Y${year}W${week}` : '—';
}
async function snap(what, sels, { small = false } = {}) {
  const id = String(n++).padStart(2, '0');
  const file = `c2-open-${id}-${what.replace(/[^\w]+/g, '-').slice(0, 40)}.png`;
  const row = { id, what, when: await clockNow(), file, t: t() };
  for (const [k, sel] of Object.entries(sels)) row[k] = await words(sel);
  row.page = await words('body');
  await page.screenshot({ path: join(OUT, file) });
  if (small) {
    await page.setViewportSize({ width: 1280, height: 720 });
    await page.waitForTimeout(400);
    row.small = file.replace('.png', '-1280.png');
    await page.screenshot({ path: join(OUT, row.small) });
    await page.setViewportSize({ width: 1440, height: 900 });
    await page.waitForTimeout(300);
  }
  rows.push(row);
  const brief = Object.entries(sels).map(([k]) => `${k} ${row[k]?.words ?? '-'}`).join(', ');
  console.log(`${row.t}s ${row.when} ${what}: ${brief}, page ${row.page?.words}`);
}
const click = async (sel, opts = {}) => {
  const el = page.locator(sel).first();
  if (!(await el.count())) return false;
  await el.click({ timeout: 1500, ...opts }).catch(() => {});
  await page.waitForTimeout(200);
  return true;
};
const visible = async (sel) => (await page.locator(sel).count()) > 0 && await page.locator(sel).first().isVisible().catch(() => false);

const SPOTS = [[0, 0], ...[1, 2, 3, 4, 5].flatMap((r) => [[r, 0], [-r, 0], [0, r], [0, -r], [r, r], [-r, -r], [r, -r], [-r, r]].map(([x, y]) => [x * 110, y * 80]))];
async function setDown() {
  const box = await page.locator('.campus-map-svg').boundingBox();
  for (const [dx, dy] of SPOTS) {
    const x = box.x + box.width / 2 + dx, y = box.y + box.height / 2 + dy - 60;
    await page.mouse.move(x, y);
    await page.waitForTimeout(90);
    if (await visible('.campus-preview.ok')) { await page.mouse.click(x, y); await page.waitForTimeout(300); return true; }
  }
  return false;
}

// Title and founding.
await snap('title screen', { dialog: '[role="dialog"][aria-modal="true"]' });
await click('.title-screen .title-primary');
await page.waitForTimeout(500);
await snap('founding form', { form: '.startup-screen, .front-screen, main' });
await page.fill('input[placeholder="e.g. Halifax"]', 'Newcomb');
await click('.startup-begin-btn');
await page.waitForTimeout(1200);

const SELS = { coach: '.opening-coach', modal: '.modal, .admissions-page, .final-page', drawer: '.objectives-popup', ticker: '.log-ticker', toasts: '.toasts' };
let speedSet = false;
let lastTicker = '';
let summerSteps = 0;
let lastClock = '', lastMove = Date.now();
while (Date.now() - started < TIMEOUT_S * 1000) {
  const now = await clockNow();
  const y = Number(now.match(/^Y(\d+)/)?.[1] ?? 0);
  if (y > YEARS) { await snap(`year ${y} begins`, SELS, { small: true }); break; }

  // The welcome letter.
  if (await visible('.modal[data-interrupt="welcome"]')) {
    await snap('welcome letter', SELS, { small: true });
    await click('.modal[data-interrupt="welcome"] .letter-actions button');
    continue;
  }
  // The admissions page (Plan 108).
  if (await visible('.admissions-page')) {
    if (await visible('.admissions-lock')) {
      await snap('admissions: the price', SELS, { small: true });
      await click('.admissions-lock');
      await page.waitForTimeout(2600);
      continue;
    }
    if (await visible('.admissions-send')) {
      await snap('admissions: the standard', SELS, { small: true });
      // Hover the handle for its label, then send.
      await click('.admissions-send');
      await page.waitForTimeout(3500);
      continue;
    }
    await page.waitForTimeout(400);
    continue;
  }
  // Any other stop: a letter, the summer's pages. Advance with its button.
  const modal = page.locator('.modal-inbox, .modal-backdrop .modal').first();
  if (await modal.count() && await modal.isVisible().catch(() => false)) {
    const kind = (await modal.getAttribute('data-interrupt')) ?? 'modal';
    const head = (await modal.locator('h2, h3').first().innerText().catch(() => '')).trim();
    const key = `${kind}:${head}:${summerSteps}`;
    if (!seen.has(key)) { seen.add(key); await snap(`stop ${kind}: ${head}`, SELS, { small: kind === 'summer' }); }
    const names = await modal.locator('button:not([disabled])').evaluateAll((els) => els.map((e) => (e.innerText || e.getAttribute('aria-label') || '').trim()));
    let pick = names.findIndex((x) => /^(continue|next|begin|done|close|ok|noted|read on|carry on|on to|see |start|→)|→$/i.test(x));
    if (pick < 0) pick = names.length - 1;
    if (kind === 'summer') summerSteps++;
    await modal.locator('button:not([disabled])').nth(pick).click({ timeout: 1500 }).catch(() => {});
    await page.waitForTimeout(600);
    lastMove = Date.now();
    continue;
  }
  // The coach card.
  if (await visible('.opening-coach')) {
    const title = await page.locator('.opening-coach').getAttribute('aria-label');
    if (!seen.has(`coach:${title}`)) { seen.add(`coach:${title}`); await snap(`coach: ${title}`, SELS, { small: true }); }
    if (await visible('.campus-map-svg.placing')) { if (!(await setDown())) console.log('no ground'); continue; }
    if (!(await click('.opening-target'))) await click('.opening-coach-actions button');
    if (await visible('.hall-offer-tile')) {
      if (!(await visible('.hall-offer-tile.selected'))) await click('.hall-offer-tile');
      await page.waitForTimeout(300);
      const k2 = 'hall-offer-picked';
      if (!seen.has(k2)) { seen.add(k2); await snap('Founders Hall: an offer picked', { ...SELS, panel: '.building-info-panel' }, { small: true }); }
      await click('.building-info-jump.opening-target');
      await click('.hall-offer-picker .building-info-jump:not([disabled])');
    }
    await page.waitForTimeout(300);
    continue;
  }

  // The drawer, open by itself or by us.
  if (await visible('.objectives-popup')) {
    const title = (await page.locator('.objectives-popup .objectives-chapter h3').first().innerText().catch(() => '')).trim() || 'between chapters';
    if (!seen.has(`drawer:${title}`)) {
      seen.add(`drawer:${title}`);
      await snap(`drawer: ${title}`, { ...SELS, briefing: '.objectives-popup .objectives-briefing', objectives: '.objectives-popup .objectives-list' }, { small: true });
      // Press the first Show me, then whatever it rings.
      if (await click('.objectives-popup .objectives-door')) {
        await page.waitForTimeout(500);
        await snap(`show me: ${title}`, { ...SELS, overlay: '.tab-overlay, .build-popup, .building-info-panel' });
        if (await click('.opening-target')) {
          await page.waitForTimeout(400);
          if (await visible('.campus-map-svg.placing')) await setDown();
          // A course row's Develop opens a drawer: take its first teacher.
          await click('.course-drawer .instructor-options button:not([disabled]), .course-drawer button.primary:not([disabled])');
        }
        await page.keyboard.press('Escape'); await page.waitForTimeout(200);
        await page.keyboard.press('Escape'); await page.waitForTimeout(200);
      } else {
        await page.keyboard.press('Escape');
      }
      continue;
    }
    await page.keyboard.press('Escape');
    await page.waitForTimeout(200);
  }

  // Act on the lead objective, as a willing newcomer would (PASSIVE=1 skips this).
  const lead = (await page.locator('.log-ticker-objective-text').innerText().catch(() => '')).trim();
  if (!process.env.PASSIVE && !(await visible('.tab-overlay')) && Date.now() - (globalThis.lastAct ?? 0) > 6000) {
    globalThis.lastAct = Date.now();
    if (/Develop|Places for|second course/.test(lead) || (globalThis.devTick = (globalThis.devTick ?? 0) + 1) % 4 === 0) {
      // Keep the committee writing: every Develop the Curriculum offers.
      await page.keyboard.press('c'); await page.waitForTimeout(400);
      for (let k = 0; k < 4; k++) { if (!(await click('.row-action-develop:not([disabled])'))) break; await page.waitForTimeout(250); }
      await page.keyboard.press('Escape'); await page.waitForTimeout(200);
    }
    const tabFor = lead.match(/Open (the )?(Treasury|Students|History|Faculty)/)?.[2];
    if (tabFor) {
      await click(`button[aria-label^="${tabFor}"]`); await page.waitForTimeout(600);
      if (!seen.has(`tab:${tabFor}`)) { seen.add(`tab:${tabFor}`); await snap(`objective opens ${tabFor}`, { ...SELS, overlay: '.tab-overlay' }); }
      await page.keyboard.press('Escape'); await page.waitForTimeout(200);
    }
  }

  // A chapter whose building waits in the build menu, ringed: site it.
  if (!(await visible('.tab-overlay')) && lead.match(/^Site a (residence|dining) hall/)) {
    if (!(await visible('.build-popup'))) await click('button:has-text("Build")');
    await page.waitForTimeout(300);
    if (await click('.build-popup .opening-target')) { await page.waitForTimeout(300); if (await visible('.campus-map-svg.placing')) await setDown(); }
    await page.keyboard.press('Escape');
  }

  // The ticker, noted when it changes.
  const tick = (await page.locator('.log-ticker').innerText().catch(() => '')).replace(/\s+/g, ' ').trim();
  const objective = (await page.locator('.log-ticker-objective').innerText().catch(() => '')).replace(/\s+/g, ' ').trim();
  if (objective && objective !== lastTicker) {
    lastTicker = objective; await snap(`ticker: ${objective.slice(0, 60)}`, SELS);
    // Between chapters: what the drawer says is coming.
    if (/^OBJECTIVES/i.test(objective) && !seen.has(`between:${objective.slice(0, 20)}`)) {
      seen.add(`between:${objective.slice(0, 20)}`);
      await click('button.log-ticker-objective');
      await page.waitForTimeout(400);
      await snap('drawer between chapters', { ...SELS, between: '.objectives-between' });
      await page.keyboard.press('Escape');
    }
  }

  // Play at double speed once the walk is done.
  if (!speedSet && await visible('button[aria-label="2×"]:not([disabled])')) {
    await click('button[aria-label="2×"]');
    speedSet = true;
  }
  // A paused clock with a notice: open it.
  if (await visible('button[aria-label="Paused"][aria-pressed="true"]') && speedSet) {
    if (await click('.toast .toast-open')) {
      await page.waitForTimeout(300);
      const subj = (await page.locator('.inbox-read h3').first().innerText().catch(() => 'a matter')).trim();
      if (!seen.has(`matter:${subj}`)) { seen.add(`matter:${subj}`); await snap(`inbox matter: ${subj}`, { ...SELS, inbox: '.tab-overlay' }); }
      await click('.inbox-read .event-choice:not([disabled])');
      await page.keyboard.press('Escape');
    }
    await click('button[aria-label="2×"]');
  }
  if (now !== lastClock) { lastClock = now; lastMove = Date.now(); }
  if (Date.now() - lastMove > 25_000) {
    await snap('STALL', SELS);
    lastMove = Date.now();
    await click('button[aria-label="2×"]');
  }
  await page.waitForTimeout(250);
}

writeFileSync(join(OUT, 'opening.json'), JSON.stringify({ rows, errors }, null, 1));
const md = ['| # | When | What | Coach | Letter/page | Drawer | Ticker | Page | Shot |', '|---:|---|---|---:|---:|---:|---:|---:|---|',
  ...rows.map((r) => `| ${r.id} | ${r.when} | ${r.what} | ${r.coach?.words ?? ''} | ${r.modal?.words ?? ''} | ${r.drawer?.words ?? ''} | ${r.ticker?.words ?? ''} | ${r.page?.words ?? ''} | ${r.file} |`)];
writeFileSync(join(OUT, 'opening.md'), md.join('\n') + '\n');
console.log(`done after ${t()}s; ${rows.length} rows; errors: ${errors.length ? errors.join('; ') : 'none'}`);
await browser.close();
